45class A2DPDecoder {
46 public:
47 A2DPDecoder(audio_tools::AudioDecoder& decoder) : p_decoder(&decoder) {
48 // decoded PCM bytes are forwarded into the sink's normal output
49 // pipeline (BluetoothA2DPSink::audio_data_callback(), via the
50 // ccall_audio_data_callback() trampoline) - so volume control, channel
51 // swap, stream_reader callbacks and write_audio() all keep working
52 // unchanged, regardless of which codec produced the PCM
53 pcm_stream.setWriteCallback(write_pcm);
54 // reacts immediately whenever the wrapped decoder reports a sample
55 // rate/channel change, instead of the sink having to poll for it
56 pcm_stream.setAudioInfoCallback(a2dp_decoder_audio_info_changed);
57 }
58 virtual ~A2DPDecoder() = default;
59
61 virtual esp_a2d_mct_t codec_type() = 0;
62
64 virtual const char* mime() = 0;
65
67 virtual void build_capability(esp_a2d_mcc_t& mcc) = 0;
68
77 virtual bool begin(const esp_a2d_mcc_t& mcc) {
78 parse_audio_info(mcc);
79 pcm_stream.setAudioInfo(
80 audio_tools::AudioInfo(sample_rate, channels, bits_per_sample));
81 p_decoder->setOutput(pcm_stream);
82 p_decoder->addNotifyAudioChange(pcm_stream);
83 return p_decoder->begin();
84 }
85
88 virtual size_t write(const uint8_t* data, size_t len) {
89 return p_decoder->write(data, len);
90 }
91
93 virtual void end() { p_decoder->end(); }
94
98 audio_tools::AudioInfo get_audio_info() { return pcm_stream.audioInfo(); }
99
100 protected:
101 audio_tools::AudioDecoder* p_decoder;
102 // doubles as the PCM output target (setOutput) and the audio-info-change
103 // notification target (addNotifyAudioChange): its write callback forwards
104 // decoded PCM out, and audioInfo() reports back whatever the wrapped
105 // decoder last reported via setAudioInfo()/notifyAudioChange()
106 audio_tools::CallbackStream pcm_stream;
107 int sample_rate = 44100;
108 int channels = 2;
109 int bits_per_sample = 16;
110
111 virtual void parse_audio_info(const esp_a2d_mcc_t& mcc) = 0;
112
113 private:
114 static size_t write_pcm(const uint8_t* data, size_t len);
115};
