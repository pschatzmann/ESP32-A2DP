61 size_t write(const uint8_t* data, size_t len) override {
62 if (len == 0) return 0;
63 size_t adts_frame_len = len + kAdtsHeaderSize;
64 if (adts_frame_len > kAdtsMaxFrameLen) {
65 // 13-bit ADTS frame-length field can't represent this frame; A2DP
66 // AAC frames are always far smaller than this in practice
67 return 0;
68 }
69 write_adts_header(adts_frame_len);
70 return A2DPDecoder::write(data, len);
71 }

73 void build_capability(esp_a2d_mcc_t& mcc) override {
74 mcc.type = ESP_A2D_MCT_M24;
75#ifdef ESP_A2D_M24_CIE_OBJ_TYPE_2_AAC_LC
76 // the named MPEG-2/4 AAC CIE bit constants (and real SEP support for
77 // this codec type) are only available starting with ESP-IDF >= 6.1
78 mcc.cie.m24_info.drc = ESP_A2D_M24_CIE_DRC_NS;
79 mcc.cie.m24_info.obj_type = ESP_A2D_M24_CIE_OBJ_TYPE_2_AAC_LC |
80 ESP_A2D_M24_CIE_OBJ_TYPE_4_AAC_LC |
81 ESP_A2D_M24_CIE_OBJ_TYPE_4_HE_AAC |
82 ESP_A2D_M24_CIE_OBJ_TYPE_4_HE_AAC_V2;
83 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_8K | ESP_A2D_M24_CIE_SF1_11K |
84 ESP_A2D_M24_CIE_SF1_12K | ESP_A2D_M24_CIE_SF1_16K |
85 ESP_A2D_M24_CIE_SF1_22K | ESP_A2D_M24_CIE_SF1_24K |
86 ESP_A2D_M24_CIE_SF1_32K | ESP_A2D_M24_CIE_SF1_44K;
87 mcc.cie.m24_info.samp_freq2 = ESP_A2D_M24_CIE_SF2_48K | ESP_A2D_M24_CIE_SF2_64K |
88 ESP_A2D_M24_CIE_SF2_88K | ESP_A2D_M24_CIE_SF2_96K;
89 mcc.cie.m24_info.ch = ESP_A2D_M24_CIE_CH_1 | ESP_A2D_M24_CIE_CH_2;
90 mcc.cie.m24_info.vbr = ESP_A2D_M24_CIE_VBR_SUPPORT;
91 mcc.cie.m24_info.br1 = 0x7F & ESP_A2D_M24_CIE_BR1_MSK;
92 mcc.cie.m24_info.br2 = 0xFF & ESP_A2D_M24_CIE_BR2_MSK;
93 mcc.cie.m24_info.br3 = 0xFF & ESP_A2D_M24_CIE_BR3_MSK;
94#else
95 // ESP-IDF < 6.1: the named bit constants aren't defined yet, and
96 // esp_a2d_sink_register_stream_endpoint() documents this codec type
97 // as unsupported anyway - advertise an empty capability rather than
98 // claim capabilities we can't back up
99#endif

112 void parse_audio_info(const esp_a2d_mcc_t& mcc) override {
113#ifdef ESP_A2D_M24_CIE_OBJ_TYPE_2_AAC_LC
114 const auto& m24 = mcc.cie.m24_info;
115 if (m24.samp_freq2 & ESP_A2D_M24_CIE_SF2_96K) {
116 sample_rate = 96000;
117 } else if (m24.samp_freq2 & ESP_A2D_M24_CIE_SF2_88K) {
118 sample_rate = 88200;
119 } else if (m24.samp_freq2 & ESP_A2D_M24_CIE_SF2_64K) {
120 sample_rate = 64000;
121 } else if (m24.samp_freq2 & ESP_A2D_M24_CIE_SF2_48K) {
122 sample_rate = 48000;
123 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_44K) {
124 sample_rate = 44100;
125 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_32K) {
126 sample_rate = 32000;
127 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_24K) {
128 sample_rate = 24000;
129 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_22K) {
130 sample_rate = 22050;
131 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_16K) {
132 sample_rate = 16000;
133 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_12K) {
134 sample_rate = 12000;
135 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_11K) {
136 sample_rate = 11025;
137 } else if (m24.samp_freq1 & ESP_A2D_M24_CIE_SF1_8K) {
138 sample_rate = 8000;
139 } else {
140 sample_rate = 44100;
141 }
142 channels = (m24.ch & ESP_A2D_M24_CIE_CH_1) ? 1 : 2;
143#else
144 // ESP-IDF < 6.1: the named bit constants aren't defined and this codec
145 // type never actually gets negotiated - keep the defaults
146#endif
147 freq_idx = adts_freq_idx(sample_rate);
148 }

152 static uint8_t adts_freq_idx(int rate) {
153 static const int table[] = {96000, 88200, 64000, 48000, 44100, 32000,
154 24000, 22050, 16000, 12000, 11025, 8000, 7350};
155 for (uint8_t i = 0; i < 13; i++) {
156 if (table[i] == rate) return i;
157 }
158 return 4; // 44100
159 }

170 void write_adts_header(size_t adts_frame_len) {
171 uint8_t out[kAdtsHeaderSize];
172 out[0] = 0xFF;
173 out[1] = 0xF1; // MPEG-4, layer 0, no CRC
174 out[2] = (1 << 6) | (freq_idx << 2) | ((channels >> 2) & 0x1); // profile=AAC-LC
175 out[3] = ((channels & 0x3) << 6) | ((adts_frame_len >> 11) & 0x3);
176 out[4] = (adts_frame_len >> 3) & 0xFF;
177 out[5] = ((adts_frame_len & 0x7) << 5) | 0x1F;
178 out[6] = 0xFC;
179 A2DPDecoder::write(out, sizeof(out));
180 }
