39class A2DPDecoderSBC : public A2DPDecoder {
40 public:
41 A2DPDecoderSBC(audio_tools::AudioDecoder& decoder) : A2DPDecoder(decoder) {}
42
43 esp_a2d_mct_t codec_type() override { return ESP_A2D_MCT_SBC; }
44
45 const char* mime() override { return "audio/sbc"; }
46
47 void build_capability(esp_a2d_mcc_t& mcc) override {
48 mcc.type = ESP_A2D_MCT_SBC;
49 mcc.cie.sbc_info.samp_freq = ESP_A2D_SBC_CIE_SF_16K | ESP_A2D_SBC_CIE_SF_32K |
50 ESP_A2D_SBC_CIE_SF_44K | ESP_A2D_SBC_CIE_SF_48K;
51 mcc.cie.sbc_info.ch_mode = ESP_A2D_SBC_CIE_CH_MODE_MONO |
52 ESP_A2D_SBC_CIE_CH_MODE_DUAL_CHANNEL |
53 ESP_A2D_SBC_CIE_CH_MODE_STEREO |
54 ESP_A2D_SBC_CIE_CH_MODE_JOINT_STEREO;
55 mcc.cie.sbc_info.block_len = ESP_A2D_SBC_CIE_BLOCK_LEN_4 | ESP_A2D_SBC_CIE_BLOCK_LEN_8 |
56 ESP_A2D_SBC_CIE_BLOCK_LEN_12 | ESP_A2D_SBC_CIE_BLOCK_LEN_16;
57 mcc.cie.sbc_info.num_subbands =
58 ESP_A2D_SBC_CIE_NUM_SUBBANDS_4 | ESP_A2D_SBC_CIE_NUM_SUBBANDS_8;
59 mcc.cie.sbc_info.alloc_mthd =
60 ESP_A2D_SBC_CIE_ALLOC_MTHD_SNR | ESP_A2D_SBC_CIE_ALLOC_MTHD_LOUDNESS;
61 mcc.cie.sbc_info.min_bitpool = 2;
62 mcc.cie.sbc_info.max_bitpool = 250;
63 }
64
65 protected:
70 void parse_audio_info(const esp_a2d_mcc_t& mcc) override {
71 const auto& sbc = mcc.cie.sbc_info;
72 if (sbc.samp_freq & ESP_A2D_SBC_CIE_SF_48K) {
73 sample_rate = 48000;
74 } else if (sbc.samp_freq & ESP_A2D_SBC_CIE_SF_44K) {
75 sample_rate = 44100;
76 } else if (sbc.samp_freq & ESP_A2D_SBC_CIE_SF_32K) {
77 sample_rate = 32000;
78 } else if (sbc.samp_freq & ESP_A2D_SBC_CIE_SF_16K) {
79 sample_rate = 16000;
80 }
81 channels = (sbc.ch_mode == ESP_A2D_SBC_CIE_CH_MODE_MONO) ? 1 : 2;
82 }
83};
