89 virtual bool begin(const esp_a2d_mcc_t& mcc) {
90 sample_rate_cfg = input_sample_rate;
91 channels_cfg = 2;
92 parse_audio_info(mcc);
93 if (sample_rate_cfg != input_sample_rate) {
94 ESP_LOGW("A2DPEncoder",
95 "negotiated sample rate %d differs from the PCM sample rate %d",
96 sample_rate_cfg, input_sample_rate);
97 }
98 configure(mcc);
99 p_encoder->setAudioInfo(
100 audio_tools::AudioInfo(sample_rate_cfg, channels_cfg, 16));
101 p_encoder->setOutput(frame_output);
102 return p_encoder->begin();
103 }
104
108 virtual size_t write(const uint8_t* pcm, size_t len) {
109 if (channels_cfg != 1) return p_encoder->write(pcm, len);
110 // stereo -> mono
111 int16_t mono[64];
112 const int16_t* stereo = (const int16_t*)pcm;
113 size_t frames = len / 4;
114 size_t pos = 0;
115 while (pos < frames) {
116 size_t n = std::min(frames - pos, sizeof(mono) / sizeof(int16_t));
117 for (size_t j = 0; j < n; j++) {
118 mono[j] = ((int32_t)stereo[(pos + j) * 2] +
119 stereo[(pos + j) * 2 + 1]) / 2;
120 }
121 p_encoder->write((const uint8_t*)mono, n * sizeof(int16_t));
122 pos += n;
123 }
124 return len;
125 }
126
128 virtual void end() { p_encoder->end(); }
129
131 void set_frame_sink(A2DPFrameSink* sink) { p_sink = sink; }

134 audio_tools::AudioInfo get_audio_info() {
135 return audio_tools::AudioInfo(sample_rate_cfg, channels_cfg, 16);
136 }

144 size_t write(const uint8_t* data, size_t len) override {
145 encoder.write_encoded(data, len);
146 return len;
147 }
