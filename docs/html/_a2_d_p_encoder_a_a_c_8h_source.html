62 void build_capability(esp_a2d_mcc_t& mcc) override {
63 mcc.type = ESP_A2D_MCT_M24;
64#ifdef ESP_A2D_M24_CIE_OBJ_TYPE_2_AAC_LC
65 mcc.cie.m24_info.drc = ESP_A2D_M24_CIE_DRC_NS;
66 mcc.cie.m24_info.obj_type =
67 ESP_A2D_M24_CIE_OBJ_TYPE_2_AAC_LC | ESP_A2D_M24_CIE_OBJ_TYPE_4_AAC_LC;
68 // we only advertise the sample rate of the provided PCM
69 mcc.cie.m24_info.samp_freq1 = 0;
70 mcc.cie.m24_info.samp_freq2 = 0;
71 switch (input_sample_rate) {
72 case 8000:
73 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_8K;
74 break;
75 case 11025:
76 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_11K;
77 break;
78 case 12000:
79 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_12K;
80 break;
81 case 16000:
82 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_16K;
83 break;
84 case 22050:
85 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_22K;
86 break;
87 case 24000:
88 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_24K;
89 break;
90 case 32000:
91 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_32K;
92 break;
93 case 48000:
94 mcc.cie.m24_info.samp_freq2 = ESP_A2D_M24_CIE_SF2_48K;
95 break;
96 default:
97 if (input_sample_rate != 44100) {
98 ESP_LOGE("A2DPEncoderAAC",
99 "unsupported sample rate %d - advertising 44100",
100 input_sample_rate);
101 }
102 mcc.cie.m24_info.samp_freq1 = ESP_A2D_M24_CIE_SF1_44K;
103 break;
104 }
105 mcc.cie.m24_info.ch = ESP_A2D_M24_CIE_CH_1 | ESP_A2D_M24_CIE_CH_2;
106 mcc.cie.m24_info.vbr = ESP_A2D_M24_CIE_VBR_NS;
107 mcc.cie.m24_info.br1 = (bitrate >> 16) & ESP_A2D_M24_CIE_BR1_MSK;
108 mcc.cie.m24_info.br2 = (bitrate >> 8) & ESP_A2D_M24_CIE_BR2_MSK;
109 mcc.cie.m24_info.br3 = bitrate & ESP_A2D_M24_CIE_BR3_MSK;
110#else
111 // ESP-IDF < 6.1: the named bit constants aren't defined yet, and AAC
112 // stream endpoints are not supported anyway
113 ESP_LOGW("A2DPEncoderAAC", "AAC requires ESP-IDF >= 6.1");
114#endif

152 static bool is_adts_sync(const uint8_t* data) {
153 return data[0] == 0xFF && (data[1] & 0xF6) == 0xF0;
154 }
155
157 void write_encoded(const uint8_t* data, size_t len) override {
158 if (len == 0) return;
159 if (mode == Undefined) {
160 mode = (len >= 2 && is_adts_sync(data)) ? ADTS : Raw;
161 }
162 if (mode == Raw) {
163 A2DPEncoder::write_encoded(data, len);
164 return;
165 }
166
167 buffer.insert(buffer.end(), data, data + len);
168 size_t pos = 0;
169 while (buffer.size() - pos >= kAdtsMinHeaderSize) {
170 const uint8_t* hdr = buffer.data() + pos;
171 if (!is_adts_sync(hdr)) {
172 // resync
173 pos++;
174 continue;
175 }
176 size_t header_size = (hdr[1] & 0x01) ? 7 : 9; // protection_absent
177 size_t frame_len = ((hdr[3] & 0x03) << 11) | (hdr[4] << 3) |
178 ((hdr[5] >> 5) & 0x07);
179 if (frame_len <= header_size) {
180 pos++;
181 continue;
182 }
183 if (buffer.size() - pos < frame_len) break; // incomplete frame
184 A2DPEncoder::write_encoded(hdr + header_size, frame_len - header_size);
185 pos += frame_len;
186 }
187 buffer.erase(buffer.begin(), buffer.begin() + pos);
188 }
